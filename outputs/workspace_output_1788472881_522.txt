# ============================================================
# 4. Bridge theme systems
# ============================================================
# The forum's setTheme function (around line 4167) uses localStorage('fedos_theme')
# We need to:
# a) On load, read fedposter_theme first, map to forum theme if needed
# b) When forum theme changes, also write to fedposter_theme
# c) Map site-only themes to closest forum theme

# Find the forum's theme initialization line:
#   let currentTheme = localStorage.getItem('fedos_theme') || 'dark';
old_theme_init = "let currentTheme = localStorage.getItem('fedos_theme') || 'dark';"
new_theme_init = """// ── Theme bridge: sync with site's fedposter_theme ──
        const FORUM_THEMES = ['dark','light','ocean','forest','sunset','midnight','rose'];
        // Map site-only themes (12-theme system) to closest forum theme (7-theme system)
        const SITE_TO_FORUM = {
            'dark':'dark','light':'light','blue':'ocean','green':'forest',
            'purple':'midnight','orange':'sunset','cyberpunk':'midnight',
            'sunset':'sunset','ocean':'ocean','rose':'rose','midnight':'midnight',
            'synthwave':'rose'
        };
        // Read site theme first, fall back to forum's own key
        let _siteTheme = localStorage.getItem('fedposter_theme') || localStorage.getItem('fedos_theme') || 'dark';
        let currentTheme = SITE_TO_FORUM[_siteTheme] || _siteTheme;
        if (!FORUM_THEMES.includes(currentTheme)) currentTheme = 'dark';"""

if old_theme_init in html:
    html = html.replace(old_theme_init, new_theme_init, 1)
    print("✅ Step 4a: Bridged theme initialization (reads fedposter_theme, maps 12→7)")
else:
    print("⚠ Step 4a: Could not find theme init line")

# Find the setTheme function and add fedposter_theme sync
old_settheme = """function setTheme(theme) {
            currentTheme = theme;
            document.documentElement.setAttribute('data-theme', theme);
            themeIcon.className = THEME_ICONS[theme] || 'fas fa-palette';
            localStorage.setItem('fedos_theme', theme);
            document.querySelectorAll('.theme-option').forEach(opt => {
                opt.classList.toggle('active', opt.dataset.themeVal === theme);
            });
        }"""

new_settheme = """function setTheme(theme) {
            if (!FORUM_THEMES.includes(theme)) theme = SITE_TO_FORUM[theme] || 'dark';
            currentTheme = theme;
            document.documentElement.setAttribute('data-theme', theme);
            themeIcon.className = THEME_ICONS[theme] || 'fas fa-palette';
            localStorage.setItem('fedos_theme', theme);
            // Sync to site theme key so other pages stay consistent
            localStorage.setItem('fedposter_theme', theme);
            document.querySelectorAll('.theme-option').forEach(opt => {
                opt.classList.toggle('active', opt.dataset.themeVal === theme);
            });
            // Update navbar theme picker swatches
            document.querySelectorAll('.nav-theme-picker .swatch').forEach(s => {
                s.classList.toggle('active', s.dataset.theme === theme);
            });
        }"""

if old_settheme in html:
    html = html.replace(old_settheme, new_settheme, 1)
    print("✅ Step 4b: Bridged setTheme (syncs to fedposter_theme, updates navbar swatches)")
else:
    print("⚠ Step 4b: Could not find setTheme function")

# Also fix the settings modal theme handler that uses fedos_theme as fallback

new_settings_theme = "else { if (typeof setTheme === 'function') setTheme(t); else { document.documentElement.setAttribute('data-theme', t); localStorage.setItem('fedos_theme', t); localStorage.setItem('fedposter_theme', t); } }"
if old_settings_theme in html:
    html = html.replace(old_settings_theme, new_settings_theme, 1)
    print("✅ Step 4c: Fixed settings modal theme fallback")

# ============================================================
# 5. Prevent fedposter.js from overriding the forum's theme
#    The site's fedposter.js calls applyTheme() on DOM ready which sets data-theme
#    from fedposter_theme. Since we now sync fedposter_theme with the forum theme,
#    this should be consistent. But fedposter.js also calls buildThemePicker()
#    which builds 12 swatches. We want the navbar to show the forum's 7 themes
#    OR the site's 12 themes. Let's let fedposter.js build its 12 swatches
#    but make clicking them map to forum themes.
#    
#    Actually, fedposter.js already builds the picker and handles clicks via
#    applyTheme(). We need applyTheme to also work with the forum.
#    Since fedposter.js sets data-theme and fedposter_theme, and our setTheme
#    reads fedposter_theme on load, this creates a nice loop:
#    - User clicks a site swatch → fedposter.js applyTheme() → sets data-theme + fedposter_theme
#    - The forum's themeObserver (line 3698) detects data-theme change → updates UI
#    But the forum theme won't match if user picks "blue" (no forum CSS for it).
#    We need to intercept. Let's add an observer that maps non-forum themes.
# ============================================================

# Find a good place to add a theme sync observer - after setTheme(currentTheme) call
old_init_call = "setTheme(currentTheme);"
new_init_call = """setTheme(currentTheme);

        // ── Sync when site navbar theme picker changes data-theme ──
        // fedposter.js may set a 12-theme value (e.g. 'blue') that the forum
        // doesn't have CSS for. This observer maps it to the closest forum theme.
        const _themeSyncObs = new MutationObserver(() => {
            const dt = document.documentElement.getAttribute('data-theme') || 'dark';
            if (!FORUM_THEMES.includes(dt)) {
                const mapped = SITE_TO_FORUM[dt] || 'dark';
                if (mapped !== currentTheme) {
                    currentTheme = mapped;
                    themeIcon.className = THEME_ICONS[mapped] || 'fas fa-palette';
                    localStorage.setItem('fedos_theme', mapped);
                    document.querySelectorAll('.theme-option').forEach(opt => {
                        opt.classList.toggle('active', opt.dataset.themeVal === mapped);
                    });
                    // Set the mapped theme so forum CSS applies
                    document.documentElement.setAttribute('data-theme', mapped);
                }
            }
        });
        _themeSyncObs.observe(document.documentElement, { attributes: true, attributeFilter: ['data-theme'] });"""

if old_init_call in html:
    html = html.replace(old_init_call, new_init_call, 1)
    print("✅ Step 5: Added theme sync observer for navbar picker")
else:
    print("⚠ Step 5: Could not find setTheme(currentTheme) call")

# ============================================================
# Write the modified file
# ============================================================
with open(forum_path, "w", encoding="utf-8") as f:
    f.write(html)

print(f"\n✅ Modified forum.html: {len(html)} chars, {html.count(chr(10))+1} lines")
